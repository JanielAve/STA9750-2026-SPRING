load_atus_data <- function(file_base = c("resp", "rost", "act")){
  if(!dir.exists(file.path("data", "mp02"))){
    dir.create(file.path("data", "mp02"), showWarnings=FALSE, recursive=TRUE)
  }
  
  file_base <- match.arg(file_base)
  
  file_name_out <- file.path("data", 
                             "mp02", 
                             glue("atus{file_base}_0324.dat"))
  
  if(!file.exists(file_name_out)){
    
    url_end <- glue("atus{file_base}-0324.zip")
    
    temp_zip <- tempfile(fileext=".zip")
    temp_dir <- tools::file_path_sans_ext(temp_zip)
    
    request("https://www.bls.gov") |>
      req_url_path("tus", "datafiles", url_end) |>
      req_headers('User-Agent' = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |>
      req_perform(temp_zip)
    
    unzip(temp_zip, exdir = temp_dir)
    
    temp_file <- file.path(temp_dir, basename(file_name_out))
    
    file.copy(temp_file, file_name_out)
  }
  
  file <- read_csv(file_name_out, 
                   show_col_types = FALSE)
  
  out <- switch(file_base, 
                
                # -----------------------------
                # RESPONDENT FILE
                # -----------------------------
                resp = file |> 
                  rename(
                    survey_weight = TUFNWGTP, 
                    time_alone = TRTALONE, 
                    survey_year = TUYEAR
                  ) |> 
                  mutate(
                    # Number of children under 18
                    num_children = TRYHHCHILD,
            
                    # Married status
                    married = TRTSPOUSE > 0,
                    
                    # Employment status
                    employed_status = case_match(TELFS,
                                                 1 ~ "Employed full-time",
                                                 2 ~ "Employed part-time",
                                                 3 ~ "Unemployed",
                                                 4 ~ "Retired",
                                                 TRUE ~ "Other"),
                    
                    # College enrollment (not high school)
                    enrolled_college = TESCHENR %in% c(1,2)
                  
                  ),
                
                # -----------------------------
                # ROSTER FILE
                # -----------------------------
                rost = file |> 
                  mutate(sex = case_match(TESEX, 1 ~ "M", 2 ~ "F")), 
                
                # -----------------------------
                # ACTIVITY FILE
                # -----------------------------
                act = file |> 
                  mutate(activity_n_of_day = TUACTIVITY_N, 
                         time_spent_min = TUACTDUR24, 
                         start_time = TUSTARTTIM,
                         stop_time = TUSTOPTIME, 
                         level1_code = paste0(TRTIER1P, "0000"), 
                         level2_code = paste0(TRTIER2P, "00"),
                         level3_code = TRCODEP)
  )
  
  out |> 
    rename(participant_id = TUCASEID) |>
    select(matches("[:lower:]", ignore.case = FALSE))
}

get_avg_hours <- function(demo_resp, act_data = act_full) {
  act_data %>%
    filter(participant_id %in% demo_resp$participant_id) %>%
    mutate(hours_spent = time_spent_min / 60) %>%
    inner_join(demo_resp %>% select(participant_id, survey_weight), by = "participant_id") %>%

    ) %>%
    arrange(desc(avg_hours))
}

compute_levels <- function(avg_hours_df) {
  avg_hours_df %>%
    mutate(
      low = avg_hours * 0.75,
      medium = avg_hours,
      high = avg_hours * 1.25,
      target = case_when(
        task_level1 %in% c("Sleeping", "Exercise/ Sports") ~ high,
        task_level1 %in% c("Childcare") ~ high,
        TRUE ~ medium
      )
    )
}

compute_fraction_hit <- function(demo_resp, level_df, act_data = act_full) {
  act_data %>%
    filter(participant_id %in% demo_resp$participant_id) %>%
    mutate(hours_spent = time_spent_min / 60) %>%
    group_by(participant_id, task_level1) %>%
    summarize(total_hours = sum(hours_spent, na.rm = TRUE), .groups = "drop") %>%
    inner_join(level_df %>% select(task_level1, target), by = "task_level1") %>%
    mutate(hits_target = total_hours >= target) %>%
    group_by(task_level1) %>%
    summarize(fraction_hit = mean(hits_target, na.rm = TRUE), .groups = "drop") %>%
    arrange(desc(fraction_hit))
}
