## Data Acquisition and Preparation cquire_ipeds_data <- function(start_year=2010, end_year=2024){ library(tidyverse) library(glue)

data_dir <- file.path("data", "mp01")

if(!dir.exists(data_dir)){
  dir.create(data_dir, showWarnings=FALSE, recursive=TRUE)
}

YEARS <- seq(start_year, end_year)

EFA_ALL <- map(YEARS, function(yy){
  if(yy <= 2022){
    ef_url <- glue("https://nces.ed.gov/ipeds/datacenter/data/EF{yy}A.zip")
    
  } else {
    ef_url <- glue("https://nces.ed.gov/ipeds/data-generator?year={yy}&tableName=EF{yy}A&HasRV=0&type=csv")
  }
  
  ef_file <- file.path(data_dir, glue("ef{yy}a.csv.zip"))
  
  if(!file.exists(ef_file)){
    message(glue("Downloading Enrollment Data for {yy} from {ef_url}"))
    download.file(ef_url, destfile = ef_file, quiet=TRUE, mode="wb")    
  }
  
  read_csv(ef_file, 
           show_col_types=FALSE) |>
    mutate(year = yy, 
           # American Indian or Alaskan Native
           enrollment_m_aian = EFAIANM, 
           enrollment_f_aian = EFAIANW, 
           # Asian
           enrollment_m_asia = EFASIAM, 
           enrollment_f_asia = EFASIAW, 
           # Black or African-American, 
           enrollment_m_bkaa = EFBKAAM, 
           enrollment_f_bkaa = EFBKAAW, 
           # Hispanic 
           enrollment_m_hisp = EFHISPM, 
           enrollment_f_hisp = EFHISPW, 
           # Native Hawaiian or Other Pacific Islander 
           enrollment_m_nhpi = EFNHPIM, 
           enrollment_f_nhpi = EFNHPIW, 
           # White
           enrollment_m_whit = EFWHITM, 
           enrollment_f_whit = EFWHITW, 
           # Two or More Races
           enrollment_m_2mor = EF2MORM, 
           enrollment_f_2mor = EF2MORW, 
           # Unknown / Undisclosed Race
           enrollment_m_unkn = EFUNKNM, 
           enrollment_f_unkn = EFUNKNW, 
           # US Non-Resident
           enrollment_m_nral = EFNRALM, 
           enrollment_f_nral = EFNRALW, 
    ) |> filter(
      (EFALEVEL %in% c(2, 12)) | (LINE %in% c(1, 15))
      # Per 2024 Data Dictionary, 
      # - EFALEVEL 2 = undergrad 
      # - EFALELVE 12 = grad
      # - Line 1 = first year first time full-time undergrad
      # - Line 15 = first year first time part-time undergrad
    ) |> mutate(level = case_when(
      EFALEVEL == 2 ~ "all undergrad", 
      EFALEVEL == 12 ~ "all graduate",
      LINE %in% c(1, 15) ~ "first year undergrad"
    )
    ) |>
    select(institution_id = UNITID, 
           year, 
           level,
           starts_with("enrollment_")) |>
    group_by(institution_id, 
             year, 
             level) |>
    summarize(across(starts_with("enrollment_"), sum), 
              .groups = "drop")
  
}) |> bind_rows()

DESC_ALL <- map(YEARS, function(yy){
  if(yy <= 2022){
    hd_url <- glue("https://nces.ed.gov/ipeds/datacenter/data/HD{yy}.zip")
    
  } else {
    hd_url <- glue("https://nces.ed.gov/ipeds/data-generator?year={yy}&tableName=HD{yy}&HasRV=0&type=csv")
  }
  
  hd_file <- file.path(data_dir, glue("hd{yy}.csv.zip"))
  
  if(!file.exists(hd_file)){
    message(glue("Downloading Institutional Descriptions for {yy} from {hd_url}"))
    download.file(hd_url, destfile = hd_file, quiet=TRUE, mode="wb")    
  }
  
  suppressWarnings(
    read_csv(hd_file, 
             show_col_types=FALSE, 
             locale=locale(encoding=if_else(yy==2024, "utf-8", "windows-1252"))) |>
      mutate(year = yy, 
             INSTNM) |> 
      select(institution_id = UNITID, 
             institution_name = INSTNM, 
             state = STABBR, 
             year)
  )
  
}) |> bind_rows()

inner_join(EFA_ALL, 
           DESC_ALL, 
           join_by(institution_id == institution_id, 
                   year == year))

}

IPEDS <- acquire_ipeds_data()

OP-ED: Intro: On June 29, 2023, The US Supreme Court changed higher education in the United States of America. The outcome of the case Students for Fair Admissions v. Harvard shifted the way admissions will be performed. The outcome of the case ended an era in which race-conscious admissions practices were executed by higher education organizations. Have our schools changed after no longer having diversity protected at the federal level. We shall analyze the admissions data from 2024 and earlier years to determine any diversity changes at Farmingdale State College and Baruch College, two different environmental settings, but both with the same results. Diversity is still of utmost importance to both our educational organizations. Our Definition of Diversity: Diversity to us is the ability to provide the opportunity of education to individuals from any background which includes any race, any ethnicity, any gender, and any sexual orientation. Having diversity helps us create a campus and environment which brings diverse viewpoints and ideas in order to grow as an educational organization. Having a diverse environment prepares us all for a world filled with different cultural backgrounds and ideas. Organization Stats: Baruch College: CUNY Baruch College is located in the greatest city on Earth, NYC, in 2024 we had a total of 3,585 Graduate students. The total number of students enrolled at Baruch was 20,081. Now for the important number which is the admissions of 2024, our first year undergrad admissions were 2,532 which are welcomed to our community. Our year to year change between 2023-2024 was from 1.2% to 2.5%, therefore we saw growth in our enrollment. Unfortunately, we did see a decrease in our incoming first year students between those years, we saw a drop of -3.1% which is not major, but we must keep monitoring in future years. Farmingdale State College: SUNY Farmingdale State College is located on Long Island in a more suburban area, at Farmingdale we had 32 graduate students, which makes sense because until recently we added graduate programs to our curriculum. The total number of students enrolled at Farmingdale State College was 10,039. The total new undergrad admissions was 1,822. Our year to year change from 2023-2024 was from 2.6% to 5.2% which is an amazing growth amount. Unlike Baruch, at Farmingdale we saw a good amount of growth between the same years when it comes to incoming first year students, the growth was 8.7%.

Demographics Changes Between 2023-2024: For Baruch College we only saw a negative change in one of our ethnic groups. That ethnic group is the Asian demographic but it was a -0.2% drop, all other groups stayed the same between both years or increased a bit. For Farmingdale State College we did see a bigger drop in one ethnic group. Unfortunately, we saw a -3.3% shift in the White demographic, but we did see increases in all other demographic ethnic groups. For long term diversity trends in both institutions we do see decreases in the White students fractions, we must understand why this is occurring. For other demographic groups we do see continued increases therefore diversity continues to be part of both our institutions despite the choices of the Supreme Court.
